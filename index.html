/* Sheet tabs: "Key" (A = question id, B = correct answer, row 1 = headers) and "Results" (row 1 = Time, Name, Score, Total, Answers). Set SHOW_SCORE to true only if players should see their score. */ const SHOW_SCORE = false; function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const data = JSON.parse(e.postData.contents); const name = String(data.name || "").trim().slice(0, 40); const answers = data.answers || {}; if (name.length < 2) return out({ error: "Enter your name." }); const ss = SpreadsheetApp.getActive(); const results = ss.getSheetByName("Results"); const last = results.getLastRow(); if (last > 1) { const names = results.getRange(2, 2, last - 1, 1).getValues().flat() .map(function (n) { return String(n).trim().toLowerCase(); }); if (names.indexOf(name.toLowerCase()) !== -1) { return out({ error: "This name has already submitted." }); } } const key = ss.getSheetByName("Key").getDataRange().getValues().slice(1) .filter(function (r) { return r[0] !== ""; }); let score = 0; key.forEach(function (r) { if (String(answers[r[0]] || "").trim() === String(r[1]).trim()) score++; }); results.appendRow([new Date(), name, score, key.length, JSON.stringify(answers)]); return SHOW_SCORE ? out({ ok: true, score: score, total: key.length }) : out({ ok: true }); } catch (err) { return out({ error: "Something went wrong. Try again." }); } finally { lock.releaseLock(); } } function out(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
